/** * セルン LINE予約 API（GAS） * スプレッドシート「セルン予約台帳」にバインドして使う */ const SHEET_RES = '予約'; const SHEET_CUS = '顧客'; const SHEET_CFG = '設定'; const HOLIDAY_CAL = 'ja.japanese#holiday@group.v.calendar.google.com'; const TZ = 'Asia/Tokyo'; /** 初回だけ実行：シートと設定を作る */ function setup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const make = (name, header) => { let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); if (sh.getLastRow() === 0) { sh.appendRow(header); sh.setFrozenRows(1); sh.getRange(1, 1, 1, header.length).setFontWeight('bold').setBackground('#f3f3f3'); } return sh; }; const res = make(SHEET_RES, ['予約ID', '受付日時', '予約日', '時間', '人数', '席希望', '名前', '電話', 'LINEユーザーID', '備考', 'ステータス']); res.getRange('B:D').setNumberFormat('@'); res.getRange('H:H').setNumberFormat('@'); res.getRange('K2:K').setDataValidation( SpreadsheetApp.newDataValidation().requireValueInList(['予約', '来店', 'キャンセル', '無断キャンセル']).build() ); const cus = make(SHEET_CUS, ['LINEユーザーID', '名前', '電話', '初回予約日', '最終予約日', '予約回数']); cus.getRange('C:E').setNumberFormat('@'); const cfg = make(SHEET_CFG, ['項目', '値', '説明']); cfg.getRange('B:B').setNumberFormat('@'); if (cfg.getLastRow() === 1) { cfg.getRange(2, 1, 8, 3).setValues([ ['受付開始', '17:00', '最初の予約枠（実際の時間に直す）'], ['最終受付', '21:00', '最後の予約枠（実際の時間に直す）'], ['刻み分', '30', '枠の間隔（分）'], ['締切日数', '2', '何日前までネット予約OKか'], ['予約可能日数', '60', '何日先まで受け付けるか'], ['最大人数', '6', 'ネット予約の上限人数'], ['休業日カレンダーID', '97535af815b73d359dbf78847f23e56026c42964dcf3581568a045bae65186d0@group.calendar.google.com', 'シフト管理と共用'], ['土日祝の案内', '土日祝はお席のご利用時間を1〜1.5時間とさせていただきます', '土日祝の予約画面に自動表示'], ]); } cfg.autoResizeColumns(1, 3); } /** 動作確認用 */ function doGet() { return json_({ ok: true, msg: 'セルンLINE予約API 稼働中' }); } /** LIFF画面からの呼び出し口 */ function doPost(e) { try { const req = JSON.parse(e.postData.contents); const user = verify_(req.idToken); // なりすまし防止：LINEのIDトークンを検証 let result; if (req.action === 'init') result = init_(user.sub); else if (req.action === 'reserve') result = reserve_(user.sub, req); else throw new Error('不明な操作です'); return json_(Object.assign({ ok: true }, result)); } catch (err) { return json_({ ok: false, error: err.message }); } } /** 画面の初期表示データ */ function init_(userId) { const cfg = getConfig_(); const c = findCustomer_(userId); return { customer: c ? { name: c.name, phone: c.phone } : null, days: calendarDays_(cfg), slots: slots_(cfg), maxPeople: cfg.maxPeople, weekendNote: cfg.weekendNote, limitMin: cfg.limitMin, freeFrom: cfg.freeFrom, }; } /** 予約登録 */ function reserve_(userId, req) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const cfg = getConfig_(); const days = calendarDays_(cfg); // 二重登録防止：同じ送信IDの予約がすでにあれば、その結果を返す const rs = sheet_(SHEET_RES); ensureHeaders_(rs); if (req.requestId && rs.getLastRow() > 1) { const v = rs.getRange(2, 1, rs.getLastRow() - 1, 12).getDisplayValues(); for (let i = v.length - 1; i >= 0; i--) { if (v[i][11] === req.requestId) { const dd = days.find(d => d.date === v[i][2]); return { reservationId: v[i][0], name: v[i][6], limited: isLimited_(dd, v[i][3], cfg), weekendNote: cfg.weekendNote, limitMin: cfg.limitMin }; } } } const day = days.find(d => d.date === req.date); if (!day) throw new Error('この日はネット予約の受付期間外です。お電話でお願いします'); if (day.closed) throw new Error('この日は休業日です'); if (slots_(cfg).indexOf(req.time) < 0) throw new Error('時間が正しくありません'); const n = Number(req.people); if (!(n >= 1 && n <= cfg.maxPeople)) throw new Error(cfg.maxPeople + '名を超える場合はお電話でお願いします'); if (['座敷', 'イス', '希望なし'].indexOf(req.seat) < 0) throw new Error('席の希望を選んでください'); // お子様の内訳（合計人数に含む） const kid = x => { const k = Math.floor(Number(x) || 0); return k > 0 ? k : 0; }; const kLow = kid(req.kidsLow), kInf = kid(req.kidsInfant), kBaby = kid(req.kidsBaby); if (kLow + kInf + kBaby > n - 1) throw new Error('お子様の人数が合計人数を超えています（大人の方1名以上）'); let c = findCustomer_(userId); const name = clean_(req.name); const phone = String(req.phone || '').replace(/[^\d]/g, ''); if (!c && (!name || phone.length < 10)) throw new Error('お名前と電話番号を入力してください'); const now = new Date(); const cs = sheet_(SHEET_CUS); if (!c) { cs.appendRow([userId, name, "'" + phone, req.date, req.date, 1]); c = { name: name, phone: phone }; } else { cs.getRange(c.row, 5, 1, 2).setValues([[req.date, c.count + 1]]); } const id = 'R' + Utilities.formatDate(now, TZ, 'yyMMddHHmmss'); rs.appendRow([ id, Utilities.formatDate(now, TZ, 'yyyy-MM-dd HH:mm'), req.date, req.time, n, req.seat, c.name, "'" + c.phone, userId, clean_(req.note).slice(0, 200), '予約', String(req.requestId || ''), kLow, kInf, kBaby, ]); notify_(cfg, { date: req.date, time: req.time, n: n, seat: req.seat, name: c.name, phone: c.phone, kids: kidsText_(kLow, kInf, kBaby), note: clean_(req.note).slice(0, 200), limited: isLimited_(day, req.time, cfg), }); return { reservationId: id, name: c.name, limited: isLimited_(day, req.time, cfg), weekendNote: cfg.weekendNote, limitMin: cfg.limitMin }; } finally { lock.releaseLock(); } } /** 予約シートの追加列の見出しを用意 */ function ensureHeaders_(rs) { const extra = ['送信ID', '小学低学年', '幼児', '赤ちゃん']; const cur = rs.getRange(1, 12, 1, 4).getValues()[0]; extra.forEach((h, i) => { if (cur[i] === '') rs.getRange(1, 12 + i).setValue(h).setFontWeight('bold').setBackground('#f3f3f3'); }); } function isLimited_(day, time, cfg) { return !!(day && day.limited && time < cfg.freeFrom); } function endTime_(t, min) { const p = t.split(':').map(Number), e = p[0] * 60 + p[1] + min; return Math.floor(e / 60) + ':' + ('0' + (e % 60)).slice(-2); } function kidsText_(low, inf, baby) { const p = []; if (low) p.push('小学低学年' + low); if (inf) p.push('幼児' + inf); if (baby) p.push('赤ちゃん' + baby); return p.length ? '（うち ' + p.join('・') + '）' : ''; } /** お店のスマホのLINEに予約通知（失敗しても予約は成立させる。失敗時はメールで代わりに通知） */ function notify_(cfg, r) { const W = ['日', '月', '火', '水', '木', '金', '土']; const d = new Date(r.date.replace(/-/g, '/')); const lines = [ '🔔 新しいネット予約', (d.getMonth() + 1) + '/' + d.getDate() + '(' + W[d.getDay()] + ') ' + r.time, r.n + '名' + r.kids + '・' + r.seat, r.name + ' 様　' + r.phone, ]; if (r.note) lines.push('ご要望：' + r.note); if (r.limited) lines.push('※土日祝 ' + r.time + '〜' + endTime_(r.time, cfg.limitMin) + '（' + (cfg.limitMin / 60) + '時間制）'); const text = lines.join('\n'); const token = PropertiesService.getScriptProperties().getProperty('LINE_ACCESS_TOKEN'); try { if (!token || !cfg.notifyTo) throw new Error('通知先またはトークン未設定'); const res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', { method: 'post', contentType: 'application/json', headers: { Authorization: 'Bearer ' + token }, payload: JSON.stringify({ to: cfg.notifyTo, messages: [{ type: 'text', text: text }] }), muteHttpExceptions: true, }); if (res.getResponseCode() !== 200) throw new Error('LINE通知エラー ' + res.getResponseCode() + ' ' + res.getContentText()); } catch (e) { try { MailApp.sendEmail(Session.getEffectiveUser().getEmail(), '【セルン予約】' + lines[1], text + '\n\n（LINE通知に失敗したためメールで送信：' + e.message + '）'); } catch (e2) { console.error('通知失敗: ' + e.message + ' / ' + e2.message); } } } /** 初回・機能追加時にエディタから1回実行：必要な権限をまとめて承認する */ function authorize() { MailApp.getRemainingDailyQuota(); UrlFetchApp.fetch('https://api.line.me/v2/bot/info', { headers: { Authorization: 'Bearer ' + PropertiesService.getScriptProperties().getProperty('LINE_ACCESS_TOKEN') }, muteHttpExceptions: true, }); CalendarApp.getDefaultCalendar(); SpreadsheetApp.getActiveSpreadsheet(); console.log('承認OK'); } /** 受付可能な日の一覧（休業日・土日祝フラグつき） */ function calendarDays_(cfg) { const today = new Date(Utilities.formatDate(new Date(), TZ, 'yyyy/MM/dd')); const from = addDays_(today, cfg.leadDays); const to = addDays_(today, cfg.rangeDays + 1); const closed = collectDates_(cfg.calId, from, to, false); const holiday = collectDates_(HOLIDAY_CAL, from, to, true); const days = []; for (let i = cfg.leadDays; i <= cfg.rangeDays; i++) { const d = addDays_(today, i); const key = fmt_(d); const dow = d.getDay(); days.push({ date: key, dow: dow, closed: closed.has(key), holiday: holiday.has(key), limited: dow === 0 || dow === 6 || holiday.has(key), }); } return days; } function collectDates_(calId, from, to, subscribeIfMissing) { let cal = CalendarApp.getCalendarById(calId); if (!cal && subscribeIfMissing) cal = CalendarApp.subscribeToCalendar(calId); if (!cal) throw new Error('カレンダーが見つかりません: ' + calId); const set = new Set(); cal.getEvents(from, to).forEach(ev => { const d = new Date(ev.getStartTime()); const end = ev.getEndTime(); while (d < end) { set.add(fmt_(d)); d.setDate(d.getDate() + 1); } }); return set; } function slots_(cfg) { const toMin = s => { const p = s.split(':').map(Number); return p[0] * 60 + p[1]; }; const pad = x => ('0' + x).slice(-2); const out = []; for (let t = toMin(cfg.start); t <= toMin(cfg.last); t += cfg.step) { out.push(pad(Math.floor(t / 60)) + ':' + pad(t % 60)); } return out; } /** LINEのIDトークン検証 → sub(ユーザーID) を返す */ function verify_(idToken) { if (!idToken) throw new Error('LINEからアクセスしてください'); const channelId = PropertiesService.getScriptProperties().getProperty('LINE_CHANNEL_ID'); if (!channelId) throw new Error('LINE_CHANNEL_IDが未設定です'); const res = UrlFetchApp.fetch('https://api.line.me/oauth2/v2.1/verify', { method: 'post', payload: { id_token: idToken, client_id: channelId }, muteHttpExceptions: true, }); if (res.getResponseCode() !== 200) throw new Error('LINE認証に失敗しました。開き直してください'); return JSON.parse(res.getContentText()); } function findCustomer_(userId) { const v = sheet_(SHEET_CUS).getDataRange().getDisplayValues(); for (let i = 1; i < v.length; i++) { if (v[i][0] === userId) return { row: i + 1, name: v[i][1], phone: v[i][2], count: Number(v[i][5]) || 0 }; } return null; } function getConfig_() { const c = {}; sheet_(SHEET_CFG).getDataRange().getDisplayValues().slice(1).forEach(r => { c[r[0]] = String(r[1]).trim(); }); return { start: c['受付開始'], last: c['最終受付'], step: Number(c['刻み分']) || 30, leadDays: Number(c['締切日数']) || 2, rangeDays: Number(c['予約可能日数']) || 60, maxPeople: Number(c['最大人数']) || 6, calId: c['休業日カレンダーID'], weekendNote: c['土日祝の案内'], notifyTo: c['通知先ユーザーID'] || '', limitMin: Number(c['土日祝の利用時間（分）']) || 90, freeFrom: c['時間制限なしの入店時刻'] || '19:00', }; } /** getSheetByNameのサイレント失敗対策：無ければ必ずエラーにする */ function sheet_(name) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); if (!sh) throw new Error('シートが見つかりません: ' + name); return sh; } function clean_(s) { s = String(s || '').trim(); return /^[=+\-@]/.test(s) ? "'" + s : s; // 数式インジェクション対策 } function addDays_(d, n) { const x = new Date(d); x.setDate(x.getDate() + n); return x; } function fmt_(d) { return Utilities.formatDate(d, TZ, 'yyyy-MM-dd'); } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
